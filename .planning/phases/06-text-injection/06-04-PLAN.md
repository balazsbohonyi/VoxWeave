    Replace the `Ok(_text) => { // Phase 6 will handle injection here. ... }` block with:
    ```rust
    Ok(text) => {
        // Show injecting state while injection runs
        indicator::show_injecting(&app_clone);

        // Extract all needed data from AppState BEFORE spawn_blocking
        // (never hold MutexGuard across thread boundary)
        let fw_info = app_clone.state::<AppState>()
            .foreground_window.lock().unwrap().clone();
        let injection_config = app_clone.state::<AppState>()
            .config.lock().unwrap().injection.clone();
        let cancel_flag = app_clone.state::<AppState>().cancel_flag.clone();

        let platform = app_clone.state::<PlatformProvider>();
        let app_for_inject = app_clone.clone();
        let platform_arc = platform.inner().clone(); // WindowsProvider is Clone via derive or manually

        tauri::async_runtime::spawn(async move {
            let fw_ref = fw_info.as_ref();
            // All injection work (including thread::sleep) must NOT block Tokio.
            // Use spawn_blocking for the blocking injection call.
            let result = {
                let platform = app_for_inject.state::<PlatformProvider>();
                let p = platform.inner();
                let text_clone = text.clone();
                let fw_clone = fw_info.clone();
                let cfg_clone = injection_config.clone();
                let cancel_clone = cancel_flag.clone();
                let p_ptr = p as *const _ as usize; // send raw ptr across spawn_blocking
                tokio::task::spawn_blocking(move || {
                    // SAFETY: PlatformProvider is 'static and lives for app lifetime
                    let p = unsafe { &*(p_ptr as *const crate::platform::PlatformProvider) };
                    crate::injection::inject_text(
                        p, p, p, p,
                        fw_clone.as_ref(),
                        &text_clone,
                        &cfg_clone,
                        cancel_clone,
                    )
                }).await.unwrap_or(Err(crate::injection::InjectionErrorPayload {
                    code: crate::injection::InjectionErrorCode::AllMethodsFailed,
                    message: "Injection task panicked".to_string(),
                    typed_chars: None,
                    total_chars: None,
                }))
            };

            match result {
                Ok(crate::injection::InjectionResult::Ok) => {
                    // Show green success flash then hide (CONTEXT.md: ~1 second)
                    indicator::show_success(&app_for_inject);
                    tokio::time::sleep(std::time::Duration::from_millis(1000)).await;
                    indicator::hide(&app_for_inject);
                }
                Ok(crate::injection::InjectionResult::CopiedToClipboard) => {
                    // Elevation dialog: user chose "Copy to clipboard".
                    // CONTEXT.md locked decision: show toast, NOT the green success flash.
                    indicator::hide(&app_for_inject);
                    let msg = serde_json::json!({
                        "type": "info",
                        "message": "Copied to clipboard — paste manually"
                    });
                    crate::transcription::show_toast_window(&app_for_inject, &msg);
                }
                Ok(crate::injection::InjectionResult::Cancelled { typed, total }) => {
                    // Show cancel toast and hide indicator
                    let payload = crate::injection::InjectionErrorPayload {
                        code: crate::injection::InjectionErrorCode::Cancelled,
                        message: format!("Cancelled — {typed} of {total} chars typed"),
                        typed_chars: Some(typed),
                        total_chars: Some(total),
                    };
                    indicator::hide(&app_for_inject);
                    crate::transcription::show_toast_window(&app_for_inject, &serde_json::to_value(&payload).unwrap_or_default());
                }
                Ok(crate::injection::InjectionResult::Err(_)) |
                Err(_) => {
                    indicator::hide(&app_for_inject);
                    if let Err(payload) = result {
                        crate::transcription::show_toast_window(&app_for_inject, &serde_json::to_value(&payload).unwrap_or_default());
                    }
                }
            }
        });
    }
    ```

    IMPORTANT: Check whether show_toast_window is pub in transcription module. If it is not pub, make it pub or replicate the eval() pattern used for transcription toasts. Also add necessary use imports at the top of hotkey/service.rs.

    NOTE: The raw pointer approach above avoids complex trait object lifetime issues. However, if PlatformProvider implements the four injection traits, prefer passing it directly via app.state() inside the spawn_blocking closure — let the compiler guide the approach. The key constraint: no MutexGuard held across await, no &dyn Trait across spawn_blocking (use concrete type or Arc<dyn Trait>).
