pub struct ForegroundWindowInfo {
    pub handle: usize,
    pub class_name: String,
    pub exe_name: String,
    pub integrity_level: u32,

pub const TERMINAL_CLASSES: &[&str] = &[
    "ConsoleWindowClass", "CASCADIA_HOSTING_WINDOW_CLASS",
    "mintty", "VirtualConsoleClass",
];

pub trait WindowInfo: Send + Sync {
    fn get_foreground_window(&self) -> Option<ForegroundWindowInfo>;
    fn restore_focus(&self, info: &ForegroundWindowInfo) -> Result<(), String>;
    fn is_terminal(&self, class_name: &str) -> bool { TERMINAL_CLASSES.contains(&class_name) }
}

pub trait ElevationChecker: Send + Sync {
    fn is_elevated(&self) -> bool;
    fn current_integrity_level(&self) -> u32;
    fn relaunch_elevated(&self) -> Result<(), String>;
}

pub trait InputSimulator: Send + Sync {
    fn send_paste(&self, class_name: &str) -> Result<(), String>;
    fn send_unicode_string(&self, text: &str) -> Result<(), String>;
    fn send_return(&self) -> Result<(), String>;
}

pub trait ClipboardAccess: Send + Sync {
    fn read_text(&self) -> Result<String, String>;
    fn write_text(&self, text: &str) -> Result<(), String>;
}

    1. Call GetForegroundWindow() — return None if hwnd.0 == 0
    2. GetClassNameW(hwnd, &mut class_buf) — decode class name from null-terminated u16 slice
    3. GetWindowThreadProcessId(hwnd, Some(&mut pid))
    4. OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, false, pid) — return None on failure
    5. QueryFullProcessImageNameW(hproc, PROCESS_NAME_WIN32, ...) — extract exe_name from last path component after '\\'
    6. CloseHandle(hproc)
    7. Call query_integrity_level(pid) for integrity_level (unwrap_or(0))
    8. Return Some(ForegroundWindowInfo { handle: hwnd.0 as usize, class_name, exe_name, integrity_level })

    restore_focus() implementation:
    1. HWND(info.handle as isize)
    2. SetForegroundWindow(hwnd) — ignore bool return (may silently fail per pitfall #1)
    3. std::thread::sleep(Duration::from_millis(50)) — allow OS to process focus change
    4. Return Ok(())

    CRITICAL: hold no MutexGuard across any of these calls. All data extracted from AppState before entering the unsafe block. Do not call these in async context — they are called from spawn_blocking or std::thread.

    InputSimulator — add a private free function `send_key_combo(vk_codes: &[VIRTUAL_KEY]) -> Result<(), String>` that builds a flat Vec<INPUT> of (key_down, key_up) pairs for each VK in order, then calls SendInput once:
