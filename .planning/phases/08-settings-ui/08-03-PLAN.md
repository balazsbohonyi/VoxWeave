<!-- HotkeyCapture design (from RESEARCH.md Pattern 3) -->
```typescript
// Key normalization map for Tauri shortcut format:
// " " → "Space", "ArrowUp" → "Up", "ArrowDown" → "Down"
// "ArrowLeft" → "Left", "ArrowRight" → "Right"
// "Control" → "Ctrl", "Meta" → "Super"
// Modifier keys: Control, Shift, Alt, Meta (do NOT trigger save on their own)
// Tab and Escape: blur widget and cancel capture (do NOT save)
// Combo format: "Ctrl+Shift+Space" — modifiers first, key last

function onKeyDown(e: KeyboardEvent) {
  e.preventDefault();
  heldKeys.add(normalizeKey(e.key));
  updatePreview();
}
function onKeyUp(e: KeyboardEvent) {
  const normalized = normalizeKey(e.key);
  if (!MODIFIERS.has(normalized)) {
    // Non-modifier released: combo is complete → save
    triggerSave(buildCombo());
    heldKeys.clear();
  } else {
    heldKeys.delete(normalized);
  }
}
```

Keyboard logic in `<script setup>`:
```typescript
const MODIFIERS = new Set(['Control', 'Shift', 'Alt', 'Meta']);
const KEY_MAP: Record<string, string> = {
  ' ': 'Space', 'ArrowUp': 'Up', 'ArrowDown': 'Down',
  'ArrowLeft': 'Left', 'ArrowRight': 'Right',
  'Control': 'Ctrl', 'Meta': 'Super',
};

const heldKeys = ref<Set<string>>(new Set());
const isCapturing = ref(false);
const currentPreview = ref('');

function normalizeKey(key: string): string {
  return KEY_MAP[key] ?? key;
}

function buildCombo(): string {
  const mods = ['Ctrl', 'Shift', 'Alt', 'Super'].filter(m => heldKeys.value.has(m));
  const nonMods = [...heldKeys.value].filter(k => !['Ctrl', 'Shift', 'Alt', 'Super'].includes(k));
  return [...mods, ...nonMods].join('+');
}

function onKeyDown(e: KeyboardEvent) {
  if (e.key === 'Escape') { isCapturing.value = false; heldKeys.value.clear(); (e.target as HTMLElement).blur(); return; }
  if (e.key === 'Tab') return; // let Tab propagate for form navigation
  e.preventDefault();
  isCapturing.value = true;
  heldKeys.value.add(normalizeKey(e.key));
  currentPreview.value = buildCombo() + (heldKeys.value.size === 1 && MODIFIERS.has(e.key) ? '+...' : '');
}

function onKeyUp(e: KeyboardEvent) {
  const normalized = normalizeKey(e.key);
  if (!MODIFIERS.has(e.key)) {
    const combo = buildCombo();
    if (combo) emit('save', combo);
    heldKeys.value.clear();
    isCapturing.value = false;
  } else {
    heldKeys.value.delete(normalized);
    currentPreview.value = buildCombo() + '+...';
  }
}
```

Silence state management:
```typescript
// Local refs for silence UI state
const lastNonZeroVadMs = ref(1500); // fallback default
const silenceEnabled = computed({
  get: () => config.value?.audio.vad_silence_ms !== 0,
  set: (val: boolean) => {
    if (!config.value) return;
    if (val) {
      // Restore last non-zero value
      saveConfig({ audio: { ...config.value.audio, vad_silence_ms: lastNonZeroVadMs.value } });
    } else {
      // Store current before zeroing
      if (config.value.audio.vad_silence_ms > 0) {
        lastNonZeroVadMs.value = config.value.audio.vad_silence_ms;
      }
      saveConfig({ audio: { ...config.value.audio, vad_silence_ms: 0 } });
    }
  }
});

const silenceSecs = computed(() =>
  ((config.value?.audio.vad_silence_ms || 1500) / 1000).toFixed(1)
);
```

Debounce the silence duration input save using a local `debounceTimer` ref:
```typescript
let debounceTimer: ReturnType<typeof setTimeout> | null = null;
function onSilenceDurationInput(e: Event) {
  if (debounceTimer) clearTimeout(debounceTimer);
  debounceTimer = setTimeout(() => {
    const val = parseFloat((e.target as HTMLInputElement).value);
    if (isNaN(val)) return;
    const ms = Math.round(Math.max(500, Math.min(5000, val * 1000)));
    lastNonZeroVadMs.value = ms;
    saveConfig({ audio: { ...config.value!.audio, vad_silence_ms: ms } });
  }, 300);
}
```
