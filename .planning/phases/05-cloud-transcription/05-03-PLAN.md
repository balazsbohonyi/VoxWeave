**Step B — Add transcribe_with_provider to service.rs (src-tauri/src/transcription/service.rs):**

This is a single-call override that reads config from AppState but substitutes the given provider, without mutating the stored config. Add after the existing transcribe_with_retry function:

```rust
/// Transcribe audio using a specific provider, overriding the configured active provider
/// for this call only. Does not mutate AppState config. Used by the fallback flow.
pub async fn transcribe_with_provider<R: Runtime>(
    app: &AppHandle<R>,
    audio: &EncodedAudio,
    target_provider: config::TranscriptionProvider,
) -> Result<String, ()> {
    // Clone config and override provider for this single invocation
    let mut call_config = {
        let state = app.state::<AppState>();
        state.config.lock().unwrap().transcription.clone()
    };
    call_config.provider = target_provider;

    // Delegate to the provider implementation directly (no retry — fallback gets one attempt)
    let provider_impl = build_provider(&call_config);
    match provider_impl.transcribe(audio).await {
        Ok(text) => {
            app.emit(TRANSCRIPTION_DONE_EVENT, &text).ok();
            Ok(text)
        }
        Err(err) => {
            // Emit error with no further fallback_provider (fallback already tried)
            let payload = TranscriptionErrorPayload {
                code: err.code,
                message: err.message,
                provider: Some(call_config.provider.to_string()),
                fallback_provider: None,
                retryable: false,
            };
            app.emit(TRANSCRIPTION_ERROR_EVENT, &payload).ok();
            Err(())
        }
    }
}
```

Note: `build_provider` is a private helper that already exists (or must exist) in service.rs from plan 02 to build the correct provider impl from config. If it is not already extracted as a named function, extract it from transcribe_with_retry now.

**Step C — Create src-tauri/src/commands/transcription.rs:**

```rust
use crate::config::TranscriptionProvider;
use crate::state::AppState;
use crate::transcription;
use tauri::{AppHandle, Runtime};

/// Re-run transcription with a fresh attempt counter.
/// Called by the frontend "Retry" button from a transcription-error event.
#[tauri::command]
pub async fn retry_transcription<R: Runtime>(app: AppHandle<R>) -> Result<(), String> {
    let encoded = {
        let state = app.state::<AppState>();
        state.last_encoded_audio.lock().unwrap().clone()
    };
    match encoded {
        None => Err("No audio available to retry.".into()),
        Some(audio) => {
            transcription::transcribe_with_retry(&app, &audio)
                .await
                .map_err(|_| "Transcription failed.".into())
        }
    }
}

/// Run transcription with a specific fallback provider (one attempt, no further fallback).
/// Called by the frontend "Try with X?" button from a transcription-error event.
/// provider: serialized TranscriptionProvider string — "openai", "groq", or "openrouter".
#[tauri::command]
pub async fn retry_transcription_with_fallback<R: Runtime>(
    app: AppHandle<R>,
    provider: String,
) -> Result<(), String> {
    let target = provider.parse::<TranscriptionProvider>()
        .map_err(|_| format!("Unknown provider: {provider}"))?;
    let encoded = {
        let state = app.state::<AppState>();
        state.last_encoded_audio.lock().unwrap().clone()
    };
    match encoded {
        None => Err("No audio available to retry.".into()),
        Some(audio) => {
            transcription::transcribe_with_provider(&app, &audio, target)
                .await
                .map_err(|_| "Fallback transcription failed.".into())
        }
    }
}
```

Note: `TranscriptionProvider` must implement `FromStr`. If it does not already, add a `FromStr` impl in config/mod.rs:

impl std::str::FromStr for TranscriptionProvider {
    type Err = ();
    fn from_str(s: &str) -> Result<Self, ()> {
        match s {
            "openai" => Ok(Self::Openai),
            "groq" => Ok(Self::Groq),
            "openrouter" => Ok(Self::Openrouter),
            "local" => Ok(Self::Local),
            _ => Err(()),
        }
    }
}

listen<TranscriptionErrorPayload>('transcription-error', (event) => {
  const payload = event.payload;

  if (payload.code === 'invalid_key') {
    // Open settings focused on transcription tab with offending provider highlighted
    invoke('open_settings_on_transcription_tab', { provider: payload.provider });
    return;

    return; // No toast for cancellation in this phase
  }

  // Build toast with action buttons
  if (payload.fallback_provider) {
    showToast({
      message: payload.message,
      action: {
        label: `Try with ${payload.fallback_provider}?`,
        // Invokes the real fallback command — uses the named provider, not the active one
        onClick: () => invoke('retry_transcription_with_fallback', { provider: payload.fallback_provider }),
      },
    });

    showToast({
      message: payload.message,
      action: {
        label: 'Retry',
        onClick: () => invoke('retry_transcription'),
      },
    });
  } else {
    showToast({ message: payload.message });
  }
});
